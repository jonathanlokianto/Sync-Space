package auth

import (
	"crypto/rand"
	"encoding/hex"
	"errors"
	"os"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

type CustomClaims struct {
	UserID    uint   `json:"user_id"`
	IsPremium bool   `json:"is_premium"`
	Role      string `json:"role"`
	jwt.RegisteredClaims
}

type TokenService interface {
	GenerateJWT(userID uint, isPremium bool, role string) (string, error)
	ValidateJWT(tokenString string) (*CustomClaims, error)
	GenerateRandomToken(length int) (string, error)
}

type jwtService struct {
	secretKey []byte
}

func NewJWTService() TokenService {
	secret := os.Getenv("JWT_SECRET")
	if secret == "" {
		secret = "default_jwt_secret_change_in_production"
	}
	return &jwtService{
		secretKey: []byte(secret),
	}
}

func (s *jwtService) GenerateJWT(userID uint, isPremium bool, role string) (string, error) {
	claims := CustomClaims{
		UserID:    userID,
		IsPremium: isPremium,
		Role:      role,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(time.Now().Add(24 * time.Hour)),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
		},
	}

	token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
	return token.SignedString(s.secretKey)
}

func (s *jwtService) ValidateJWT(tokenString string) (*CustomClaims, error) {
	token, err := jwt.ParseWithClaims(tokenString, &CustomClaims{}, 
		func(token *jwt.Token) (interface{}, error) {
			if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
				return nil, errors.New("unexpected signing method")
			}
			return s.secretKey, nil
	})

	if err != nil || !token.Valid {
		return nil, errors.New("invalid or expired token")
	}

	claims, ok := token.Claims.(*CustomClaims)
	if !ok {
		return nil, errors.New("failed to parse claims")
	}

	return claims, nil
}

func (s *jwtService) GenerateRandomToken(length int) (string, error) {
	bytes := make([]byte, length)
	if _, err := rand.Read(bytes); err != nil {
		return "", err
	}
	return hex.EncodeToString(bytes), nil
}

	RequireRole(expectedRole string) gin.HandlerFunc

type middlewareManager struct {
	tokenService TokenService
}

func NewMiddlewareManager(tokenService TokenService) MiddlewareManager {
	return &middlewareManager{tokenService: tokenService}
}


func (m *middlewareManager) AuthMiddleware() gin.HandlerFunc {
	return func(c *gin.Context) {
		authHeader := c.GetHeader("Authorization")
		if authHeader == "" {
			// c.JSON(http.StatusUnauthorized, gin.H{"error": "Token Not Found"})
			// c.Abort()
			c.Next()
			return
		}

		tokenString := strings.Replace(authHeader, "Bearer ", "", 1)
		
		claims, err := m.tokenService.ValidateJWT(tokenString)
		if err != nil {
			c.JSON(http.StatusUnauthorized, gin.H{"error": "Token Invalid"})
			c.Abort()
			return
		}

		c.Set("user_id", claims.UserID)
		c.Set("role", claims.Role)
		c.Set("is_premium", claims.IsPremium)

		c.Next()
	}
}

package auth

import (
	"crypto/hmac"
	"crypto/sha256"
	"os"

	"golang.org/x/crypto/bcrypt"
)


func getPepper() string {
	pepper := os.Getenv("APP_PEPPER")
	return pepper
}


func pepperPassword(password string) []byte {
	mac := hmac.New(sha256.New, []byte(getPepper()))
	mac.Write([]byte(password))
	return mac.Sum(nil)
}

func HashPassword(password string) (string, error) {
	pepperedBytes := pepperPassword(password)
	hashedBytes, error := bcrypt.GenerateFromPassword(pepperedBytes, bcrypt.DefaultCost)
	if error != nil {
		return "", error
	}

	return string(hashedBytes), nil
}


func CheckPasswordHash(inputPassword string, savedPasswordHash string) bool {
	peppereredBytes := pepperPassword(inputPassword)
	err:= bcrypt.CompareHashAndPassword([]byte(savedPasswordHash), peppereredBytes)
	return err == nil
}


func HasAndPepperPassword(password string) (string, error) {
	passwordBytes := []byte(password + "APPLICATION_PEPPER")
	hashedPassword, err := bcrypt.GenerateFromPassword(
		passwordBytes,
		bcrypt.DefaultCost,
	)

	if err != nil {
		return "", err
	}

	return string(hashedPassword), nil
}

package auth

import (
	"context"
	"errors"
	"sync-space-backend/internal/users"
)

type Service interface {
	RegisterUser(ctx context.Context, req RegisterRequest) (*RegisterResponse, error)
	LoginUser(ctx context.Context, req LoginRequest) (*LoginResponse, error)
}

type service struct {
	repository users.Repository
}

func NewService(repository users.Repository) Service {
	return &service{repository: repository}
}

func (s *service) RegisterUser(ctx context.Context, req RegisterRequest) (*RegisterResponse, error) {
	existingUser, _ := s.repository.FindByEmail(req.Email)
	if existingUser.ID != 0 {
		return nil, errors.New("Email already exist.")
	}

	hashedPassword, err := HashPassword(req.Password)
	if err != nil {
		return nil, err
	}

	userEntity := users.User{
		Fullname:     req.Fullname,
		Email:        req.Email,
		Username:     req.Username,
		Password:     string(hashedPassword),
		MaxRoomLimit: 3,
		IsPremium:    false,
	}

	err = s.repository.CreateUser(ctx, &userEntity)
	if err != nil {
		return nil, err
	}

	response := RegisterResponse{
		ID:           userEntity.ID,
		Fullname:     userEntity.Fullname,
		Username:     userEntity.Username,
		Role:         "user",
		MaxRoomLimit: userEntity.MaxRoomLimit,
		IsPremium:    userEntity.IsPremium,
	}

	return &response, nil
}

func (s *service) LoginUser(ctx context.Context, req LoginRequest) (*LoginResponse, error) {
	existingUserIdentity, _ := s.repository.FindByIdentity(req.Identity)
	if existingUserIdentity.ID == 0 {
		return nil, errors.New("No user found")
	}

	isUserValid := CheckPasswordHash(req.Password, existingUserIdentity.Password)

	if !isUserValid{
		return nil, errors.New("Email or password is incorrect.")
	}

	response := LoginResponse{
		ID:           existingUserIdentity.ID,
		Fullname:     existingUserIdentity.Fullname,
		Username:     existingUserIdentity.Username,
		Role:         "user",
		MaxRoomLimit: existingUserIdentity.MaxRoomLimit,
		IsPremium:    existingUserIdentity.IsPremium,
	}

	return &response, nil
}

func RunMigration(db *gorm.DB) error {
	query := `
		CREATE TABLE IF NOT EXISTS users (
				id SERIAL PRIMARY KEY,
				fullname VARCHAR(255) NOT NULL,
				email VARCHAR(255) UNIQUE NOT NULL,
				username VARCHAR(255) UNIQUE NOT NULL,
				password VARCHAR(255) NOT NULL,
				max_rooms_limit INT DEFAULT 3,
				is_premium BOOLEAN DEFAULT true,
				created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
				updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
		);

		CREATE TABLE IF NOT EXISTS rooms (
				id BIGSERIAL PRIMARY KEY,
				user_id INT REFERENCES users(id) ON DELETE SET NULL,
				name  VARCHAR(100) NOT NULL,
				slug  VARCHAR(100) UNIQUE NOT NULL,
				content TEXT,
				created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
				updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
		);

		CREATE INDEX IF NOT EXISTS idx_rooms_user_id ON rooms (user_id);
	`

	if err := db.Exec(query).Error; err != nil {
		return fmt.Errorf("Failure in running database migration: %w", err)
	}
	return nil
}

func (s *service) generateUniqueRandomSlug() (string, error) {
	maxRetries := 20
	for i:= 0 ; i < maxRetries; i++ {
		slugString := generator.GenerateRandomString(10)
		existingRoom, err := s.repository.FindRoomBySlug(slugString)
		if err != nil {
			if errors.Is(err, gorm.ErrRecordNotFound){
				return slugString, nil
			}
			return "", err
		}

		if existingRoom == nil {
			return slugString, nil
		}
	}
	return "", errors.New("Failed to generate unique slug. Please try again!")
}

		Slug: slug,
	}



	if err := s.repository.CreateRoom(ctx, &roomEntity, userID); err != nil {

	response := RoomResponse{

type Server struct {
	Router *gin.Engine
	DB  *gorm.DB
	AuthHandler *auth.Handler
	UserHandler *users.Handler
	RoomHandler *rooms.Handler
	Middleware  auth.MiddlewareManager
}

func NewServer(
		db *gorm.DB, 
		authHandler *auth.Handler, 
		userHandler *users.Handler, 
		roomHandler *rooms.Handler,
		mw auth.MiddlewareManager,
	) *Server {
		router := gin.Default()
		router.Use(middleware.CORSMiddleware())
		server := &Server{
			Router:      router,
			DB:          db,
			AuthHandler: authHandler,
			UserHandler: userHandler,
			RoomHandler: roomHandler,
			Middleware:  mw,
		}
		server.RegisterRoutes()
		return server
	}

func (s *Server) Start(port string) error {
	srv := &http.Server{
		Addr: fmt.Sprintf(":%s", port),
		Handler: s.Router,
		ReadTimeout: 10 * time.Second,
		WriteTimeout: 10 * time.Second,
	}

	shutdownChan := make(chan os.Signal, 1)
	signal.Notify(shutdownChan, os.Interrupt, syscall.SIGTERM)

	go func() {
		log.Printf("gin server is running on port %s", port)
		if err:= srv.ListenAndServe() ; 
			err != nil && !errors.Is(err, http.ErrServerClosed){
				log.Fatalf("Failed in running the server: %v", err)
			}
	}()

	<-shutdownChan
	log.Println("Shutdown signal received...")
	ctx, cancel := context.WithTimeout(context.Background(), 10 * time.Second)
	defer cancel()

	if err := srv.Shutdown(ctx); err != nil {
		return fmt.Errorf("Server force shutdown error: %w", err)
	}
	log.Println("Server stopped")
	return nil
}

	})

package generator

import (
	"crypto/rand"
	"math/big"
)

const charset = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"

func GenerateRandomString(n int) string {
	b := make([]byte, n)
	for i := range b {
		num, _ := rand.Int(rand.Reader,big.NewInt(int64(len(charset))))
		b[i] = charset[num.Int64()]
	}
	return string(b)
}

type CustomValidator struct {
	validator *validator.Validate
}

var slugRegex = regexp.MustCompile("^[a-z0-9]+(?:-[a-z0-9]+)*$")

func NewCustomValidator() *CustomValidator {
	v := validator.New()

	v.RegisterValidation("slug", func(fl validator.FieldLevel) bool {
		return true 
	})

	return &CustomValidator{validator: v}
}

func (cv *CustomValidator) ValidateStruct(s interface{}) map[string]string {
	err := cv.validator.Struct(s)
	if err == nil {
		return nil
	}

	errorMessages := make(map[string]string)
	for _, e := range err.(validator.ValidationErrors) {
		switch e.Tag() {
		case "required":
			errorMessages[e.Field()] = fmt.Sprintf("Field %s wajib diisi", e.Field())
		case "min":
			errorMessages[e.Field()] = fmt.Sprintf("Field %s minimal %s karakter", e.Field(), e.Param())
		default:
			errorMessages[e.Field()] = "Format tidak valid"
		}
	}

	return errorMessages
}
